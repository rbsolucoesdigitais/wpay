const NOME_ABA = "Página1";

// Defina aqui as senhas e os respectivos usuários de forma segura no servidor
const USUARIOS_AUTORIZADOS = {
  "@Rib3284": "Ribeiro",
  "#4J578e29": "Jussara"
};


function doGet() {
  return respostaJSON({ status: "erro", mensagem: "Acesso não autorizado." });
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const senhaInformada = payload.senha;

    if (!senhaInformada || !USUARIOS_AUTORIZADOS[senhaInformada]) {
      return respostaJSON({ status: "erro", mensagem: "Senha incorreta ou acesso negado." });
    }

    const usuarioIdentificado = USUARIOS_AUTORIZADOS[senhaInformada];

    if (payload.acao === "validar_senha") {
      return respostaJSON({
        status: "sucesso",
        mensagem: "Acesso liberado!",
        usuario: usuarioIdentificado
      });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(NOME_ABA) || ss.getSheets()[0];

    // Ação: Listar Contas
    if (payload.acao === "listar") {
      const dados = sheet.getDataRange().getValues();
      if (dados.length <= 1) {
        return respostaJSON({ status: "sucesso", dados: [] });
      }

      const registros = [];
      for (let i = 1; i < dados.length; i++) {
        const linha = dados[i];
        if (!linha[0]) continue;

        registros.push({
          id: linha[0],
          usuario: linha[1],
          conta: linha[2],
          vencimento: linha[3],
          valor: linha[4],
          descricao: linha[5],
          criadoEm: linha[6],
          status: linha[7] === "Pago" ? "Pago" : "Pendente"
        });
      }

      return respostaJSON({ status: "sucesso", dados: registros });
    }

    // Ação: Alternar Status (Pago / Pendente)
    if (payload.acao === "alternar_status") {
      const dados = sheet.getDataRange().getValues();
      for (let i = 1; i < dados.length; i++) {
        if (String(dados[i][0]) === String(payload.id)) {
          const numeroLinha = i + 1;
          const novoStatus = payload.status === "Pago" ? "Pago" : "Pendente";
          sheet.getRange(numeroLinha, 8).setValue(novoStatus);
          return respostaJSON({
            status: "sucesso",
            mensagem: `Conta marcada como ${novoStatus}!`,
            novoStatus: novoStatus
          });
        }
      }
      return respostaJSON({ status: "erro", mensagem: "Conta não encontrada." });
    }

    // Ação: Excluir Conta
    if (payload.acao === "excluir") {
      const dados = sheet.getDataRange().getValues();
      for (let i = 1; i < dados.length; i++) {
        if (String(dados[i][0]) === String(payload.id)) {
          sheet.deleteRow(i + 1);
          return respostaJSON({ status: "sucesso", mensagem: "Conta excluída com sucesso!" });
        }
      }
      return respostaJSON({ status: "erro", mensagem: "Conta não encontrada para exclusão." });
    }

    // Ação: Editar Conta
    if (payload.acao === "editar") {
      const dados = sheet.getDataRange().getValues();
      for (let i = 1; i < dados.length; i++) {
        if (String(dados[i][0]) === String(payload.id)) {
          const numeroLinha = i + 1;
          sheet.getRange(numeroLinha, 3).setValue(payload.conta || "");
          sheet.getRange(numeroLinha, 4).setValue(payload.vencimento || "");
          sheet.getRange(numeroLinha, 5).setValue(Number(payload.valor) || 0);
          sheet.getRange(numeroLinha, 6).setValue(payload.descricao || "");
          return respostaJSON({ status: "sucesso", mensagem: "Conta atualizada com sucesso!" });
        }
      }
      return respostaJSON({ status: "erro", mensagem: "Conta não encontrada para edição." });
    }

    // Ação Padrão: Cadastrar Nova Conta
    const novaLinha = [
      payload.id || Date.now(),
      usuarioIdentificado,
      payload.conta || "",
      payload.vencimento || "",
      Number(payload.valor) || 0,
      payload.descricao || "",
      payload.criadoEm || new Date().toISOString(),
      "Pendente"
    ];

    sheet.appendRow(novaLinha);

    return respostaJSON({
      status: "sucesso",
      mensagem: "Registro salvo com sucesso!",
      usuario: usuarioIdentificado
    });

  } catch (erro) {
    return respostaJSON({ status: "erro", mensagem: erro.toString() });
  }
}

function respostaJSON(objeto) {
  return ContentService
    .createTextOutput(JSON.stringify(objeto))
    .setMimeType(ContentService.MimeType.JSON);
}
